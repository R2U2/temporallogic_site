Previous model and add the ability to turn the rover

    
        MODULE main
        VAR
        -- Declare set of system variables
        pp30 : boolean;
        pr30 : boolean;
        bar : boolean;
        drop : boolean;
        pin : boolean;
        forward : boolean;
        reverse : boolean;
        tr : boolean;
        tl : boolean;
        sample : boolean;

        INIT
        -- Initially, all states are false/off
        pp30 = FALSE &
        pr30 = FALSE &
        bar = FALSE &
        drop = FALSE &
        pin = FALSE &
        forward = FALSE &
        reverse = FALSE &
        tr = FALSE &
        tl = FALSE &
        sample = FALSE;

        TRANS

        -- State 0 -> Rover is sitting still
        ((!pp30 & !pr30 & !bar & !drop & !pin & !forward & !reverse & !tr & !tl & !sample) ->
        -- State 0: Nothing changes.
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(!forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 1: Rover begins to move forward.
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 6: The rover stops to collect a sample.
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(!forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(sample))) |
        -- State 7: The rover is moving forward and is turning
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(forward)) & (next(!reverse)) & (next(tr)) | (next(tl)) & (next(!sample)))
        ) &

        -- State 1 -> Rover is moving forward and all is well
        ((!pp30 & !pr30 & !bar & !drop & !pin & forward & !reverse & !tr & !tl & !sample) ->
        -- State 0: Rover stops moving forward, no violations are detected.
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(!forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 1: Nothing changes.
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 2: Sensor detects anything obstacle.
        ((next(pp30)) | (next(pr30)) | (next(bar)) | (next(drop)) & (next(!pin)) & (next(forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 7: The rover is moving forward and is turning
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(forward)) & (next(!reverse)) & (next(tr)) | (next(tl)) & (next(!sample)))
        ) &

        -- State 2 -> Sensor system detects obstacle
        ((pp30 | pr30 | bar | drop & !pin & forward & !reverse & !tr & !tl & !sample) ->
        -- State 3: Stop pin is pressed by sensor system.
        ((next(pp30)) | (next(pr30)) | (next(bar)) | (next(drop)) & (next(pin)) & (next(!forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample)))
        ) &

        -- State 3 -> Stop pin is pressed and the rover has stopped
        ((pp30 | pr30 | bar | drop & pin & !forward & !reverse & !tr & !tl & !sample) ->
        -- State 4: The rover has detected an obstacle and is reversing.
        ((next(pp30)) | (next(pr30)) | (next(bar)) | (next(drop)) & (next(pin)) & (next(!forward)) & (next(reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample)))
        ) &

        -- State 4 -> Rover is reversing
        ((pp30 | pr30 | bar | drop & pin & !forward & reverse & !tr & !tl & !sample) ->
        -- State 4: Rover keeps reversing.
        ((next(pp30)) | (next(pr30)) | (next(bar)) | (next(drop)) & (next(pin)) & (next(!forward)) & (next(reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 5: Sensor detects that the obstacle has been cleared.
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(pin)) & (next(!forward)) & (next(reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample)))
        ) &

        -- State 5 -> Obstacle has been cleared
        ((!pp30 & !pr30 & !bar & !drop & pin & !forward & reverse & !tr & !tl & !sample) ->
        -- State 0: Rover is stopped and stop pin is no longer pressed
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(!forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample)))
        ) &

        -- State 6 -> The rover stops to collect a sample
        ((!pp30 & !pr30 & !bar & !drop & !pin & !forward & !reverse & !tr & !tl & sample) ->
        -- State 0 -> The rover has finished collecting its sample and is stationary
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(!forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample)))
        ) &

        -- State 7 -> The rover is moving forward and is turning
        ((!pp30 & !pr30 & !bar & !drop & !pin & forward & !reverse & tr | tl & !sample) ->
        -- State 0: Rover stops moving forward, no violations are detected.
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(!forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 1: The rover continues to move forward but does not turn
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(forward)) & (next(!reverse)) & (next(!tr)) & (next(!tl)) & (next(!sample))) |
        -- State 7: The rover continues to move forward and is turning
        ((next(!pp30)) & (next(!pr30)) & (next(!bar)) & (next(!drop)) & (next(!pin)) & (next(forward)) & (next(!reverse)) & (next(tr)) | (next(tl)) & (next(!sample)))
        )

        LTLSPEC !(G((pp30 | pr30) -> F(pin)))
        LTLSPEC !(G((!pp30 & !pr30) -> !pin))
        LTLSPEC !(G(bar -> F(pin)))
        LTLSPEC !(G(!bar -> !pin))
        LTLSPEC !(G(drop -> F(pin)))
        LTLSPEC !(G(!drop -> !pin))
        LTLSPEC !(G(!(sample & (forward | reverse))))
        LTLSPEC !(G(!(pin & (tr | tl))))
